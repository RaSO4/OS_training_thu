# ---------- 探针: 亲和性稳定性 + 每核 VLEN (rv-bench probe 模式) ----------
# 钉核后连续采样 (sched_getcpu, vlenb) 2,000,000 次的跳变计数
PROBE pinned=0  cpu=0  vlenb=32  (VLEN 256)  transitions=0   # X100
PROBE pinned=9  cpu=9  vlenb=128 (VLEN 1024) transitions=0   # A100
# 标量依赖循环 5M 迭代的 time CSR 增量 (两簇几乎相同 => 标量性能相近):
PROBE_TIME_DELTA 195665 ticks pin=0  cpu=0    # X100
PROBE_TIME_DELTA 200324 ticks pin=9  cpu=9    # A100

# ---------- 结论 ----------
# 1. sched_setaffinity 在 StarryOS 上真实有效 (2M 采样零跳变), 可把任务
#    钉到任意 X100/A100 核, 无需改内核
# 2. A100 的普通 RVV 吞吐反而比 X100 低 ~5 倍 (0.48-0.82 vs 3.7-4.3 GFLOPS),
#    而标量性能相近 => A100 的价值必须通过 IME 自定义指令
#    (llama.cpp GGML_CPU_RISCV64_SPACEMIT 路径, vmadot/vfwmadot 等) 发挥,
#    这决定了后续"负载均衡"的正确方向: X100 跑通用向量代码, A100 走 IME 算子
# 3. OPP 表: X100 上限 2.4GHz / A100 2.0GHz, 20% 的频率差解释不了 5 倍差距
